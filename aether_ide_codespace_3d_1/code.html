import { Engine } from '@aether/core';
import { FluidShader } from '@aether/materials';

// Initialize spatial engine
const engine = new Engine({
  antialias: true,
  alpha: true
});

const shader = new FluidShader({
  baseColor: '#0b1326',
  accentColor: '#adc6ff',
  density: 0.8
});

// selection highlight simulation
engine.scene.add(shader);
engine.start();

function update() {
  shader.time += 0.01;
  requestAnimationFrame(update);
}
